Google Apps Script Backend Setup

GOOGLE APPS SCRIPT DEPLOYMENT INSTRUCTIONS:

  1. In your Google Sheet, click Extensions > Apps Script and paste the script code below.
  2. Click Deploy > New deployment.
  3. Select type: Web app.
  4. Execute as: Me.
  5. Who has access: Anyone.
function doGet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var logs = [];
  var members = ["Jasper", "Ezinne", "Chinaah"];
  
  if (data.length > 1) {
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      if (row[0]) {
        logs.push({
          id: String(row[0]),
          date: String(row[1]),
          member: String(row[2]),
          startTime: String(row[3]),
          endTime: String(row[4]),
          hours: String(row[5]),
          rate: String(row[6]),
          project: String(row[7] || "Client Project"),
          notes: String(row[8] || "")
        });
        if (members.indexOf(row[2]) === -1) members.push(String(row[2]));
      }
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    logs: logs,
    members: members
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var contents = JSON.parse(e.postData.contents);
  var action = contents.action;
  
  if (action === "addShift") {
    var log = contents.log;
    sheet.appendRow([log.id, log.date, log.member, log.startTime, log.endTime, log.hours, log.rate, log.project || "Client Project", log.notes]);
    return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
  } else if (action === "deleteShift") {
    var idToDelete = String(contents.id);
    var data = sheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]) === idToDelete) {
        sheet.deleteRow(i + 1);
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "deleted" })).setMimeType(ContentService.MimeType.JSON);
  }
}
